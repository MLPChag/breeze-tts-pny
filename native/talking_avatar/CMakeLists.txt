cmake_minimum_required(VERSION 3.20)
project(talking_avatar LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options("-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=native/talking_avatar"
                        "-ffile-prefix-map=${CMAKE_BINARY_DIR}=build/native")
    if(DEFINED GODOT_CPP)
        add_compile_options("-ffile-prefix-map=${GODOT_CPP}=vendor/godot-cpp")
    endif()
endif()
find_package(Threads REQUIRED)
if(TALKING_AVATAR_JSON_INCLUDE)
    add_library(ta_json INTERFACE)
    target_include_directories(ta_json INTERFACE "${TALKING_AVATAR_JSON_INCLUDE}")
    add_library(nlohmann_json::nlohmann_json ALIAS ta_json)
else()
    find_package(nlohmann_json REQUIRED)
endif()
add_library(ta_core STATIC src/core.cpp)
target_link_libraries(ta_core PUBLIC nlohmann_json::nlohmann_json)
if(MSVC)
    target_compile_options(ta_core PRIVATE /O2 /fp:strict /W4)
else()
    target_compile_options(ta_core PRIVATE -O3 -ffp-contract=off -Wall -Wextra)
endif()
add_executable(ta_cli src/cli.cpp)
target_link_libraries(ta_cli PRIVATE ta_core)
if(DEFINED GODOT_CPP)
    set(GODOTCPP_DISABLE_EXCEPTIONS OFF CACHE BOOL "" FORCE)
    set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
    set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE PATH "" FORCE)
    add_subdirectory("${GODOT_CPP}" godot-cpp)
    if(MINGW AND CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
        # The pinned 4.5 bindings rely on transitive stdlib includes for realloc/free.
        target_compile_options(godot-cpp PRIVATE -include cstdlib)
    endif()
    add_library(talking_avatar SHARED src/analyzer.cpp src/register.cpp)
    target_link_libraries(talking_avatar PRIVATE ta_core godot::cpp Threads::Threads)
    if(UNIX AND NOT APPLE)
        target_link_options(talking_avatar PRIVATE -Wl,-Bsymbolic)
    elseif(MINGW)
        target_link_options(talking_avatar PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    set_target_properties(talking_avatar PROPERTIES
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
